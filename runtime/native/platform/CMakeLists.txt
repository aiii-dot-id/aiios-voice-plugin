cmake_minimum_required(VERSION 3.22)
project(aii_native_paths LANGUAGES CXX)
enable_testing()
add_executable(aii_native_paths_test paths_test.cpp)
target_compile_features(aii_native_paths_test PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(aii_native_paths_test PRIVATE /W4 /WX /EHsc)
  set_property(TARGET aii_native_paths_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_compile_options(aii_native_paths_test PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME native_physical_paths COMMAND aii_native_paths_test --self-test)
