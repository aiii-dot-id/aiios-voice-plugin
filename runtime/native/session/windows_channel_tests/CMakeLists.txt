cmake_minimum_required(VERSION 3.20)
project(aii_model_channel_windows CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SESSION "${CMAKE_CURRENT_SOURCE_DIR}/..")
add_library(aii_model_channel STATIC ${SESSION}/model_channel.cpp ${SESSION}/model_link.cpp ${SESSION}/model_server_control.cpp ${SESSION}/admission.cpp)
target_include_directories(aii_model_channel PUBLIC ${SESSION})
target_compile_options(aii_model_channel PRIVATE /W4 /WX /EHsc)
add_executable(aii_model_channel_windows_test ${SESSION}/model_channel_windows_test.cpp)
target_link_libraries(aii_model_channel_windows_test PRIVATE aii_model_channel)
target_compile_options(aii_model_channel_windows_test PRIVATE /EHsc)
add_executable(aii_worker_dumps_test ${SESSION}/worker_dumps_test.cpp ${SESSION}/worker_dumps.cpp)
target_link_libraries(aii_worker_dumps_test PRIVATE wer)
target_compile_options(aii_worker_dumps_test PRIVATE /EHsc)
