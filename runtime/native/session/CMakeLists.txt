option(AII_SESSION_REAL_PROBE "Build explicitly bound real-model composition probe" OFF)
if(AII_SESSION_REAL_PROBE)
  include("${CMAKE_CURRENT_LIST_DIR}/require_asr.cmake")
  # An explicitly qualified shared backend is the portable link contract.
  # The retained Mac archive recipe remains available unchanged below.
  if(DEFINED AII_POCKET_LIBRARY)
    if(NOT EXISTS "${AII_POCKET_LIBRARY}")
      message(FATAL_ERROR "Existing bound AII_POCKET_LIBRARY required")
    endif()
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
      include("${CMAKE_CURRENT_LIST_DIR}/require_pocket.cmake")
    endif()
    add_library(aii_pocket_session INTERFACE)
    target_link_libraries(aii_pocket_session INTERFACE "${AII_POCKET_LIBRARY}")
  else()
  foreach(required AII_COMPONENT_BUILD POCKET_SOURCE POCKET_BUILD)
    if(NOT DEFINED ${required} OR NOT EXISTS "${${required}}")
      message(FATAL_ERROR "Existing, bound ${required} required")
    endif()
  endforeach()
  if(NOT APPLE)
    message(FATAL_ERROR "Non-Apple builds require explicit qualified component libraries")
  endif()
  # Reuse the retained native implementation and libraries, never a Python
  # launcher or a system inference dependency. Other link recipes are explicit
  # subsequent qualifications; the common state machine does not change.
  add_library(aii_pocket_session STATIC ../../native_pocket/resident.cpp ../../native_pocket/assets_bound.cpp)
  target_compile_features(aii_pocket_session PRIVATE cxx_std_17)
  target_include_directories(aii_pocket_session PRIVATE "${POCKET_SOURCE}/include"
    "${POCKET_SOURCE}/external/ggml/include" "${POCKET_SOURCE}/src/models/pocket_tts")
  target_link_libraries(aii_pocket_session PUBLIC
    "${POCKET_BUILD}/libengine_runtime.a" "${POCKET_BUILD}/ggml/src/libggml.a"
    "${POCKET_BUILD}/ggml/src/libggml-cpu.a" "${POCKET_BUILD}/ggml/src/ggml-blas/libggml-blas.a"
    "${POCKET_BUILD}/ggml/src/libggml-base.a" "-framework Accelerate"
    "${POCKET_BUILD}/external/sentencepiece/src/libsentencepiece.a"
    "${POCKET_BUILD}/libcjson_vendor.a" "${POCKET_BUILD}/libyaml_vendor.a")
  if(DEFINED AII_POCKET_HISTORY_SOURCE_DIR)
    # Private, hash-bound overlay: never edit the retained upstream checkout or
    # the previously qualified checkpoint libraries. Objects precede the archive.
    add_library(aii_pocket_history OBJECT
      "${AII_POCKET_HISTORY_SOURCE_DIR}/src/models/pocket_tts/session.cpp"
      "${AII_POCKET_HISTORY_SOURCE_DIR}/src/models/pocket_tts/acoustic_model.cpp")
    target_compile_features(aii_pocket_history PRIVATE cxx_std_17)
    target_compile_definitions(aii_pocket_history PRIVATE GGML_GELU_FORCE_F32 GGML_USE_BLAS GGML_USE_CPU)
    target_include_directories(aii_pocket_history PRIVATE "${POCKET_SOURCE}/include"
      "${POCKET_SOURCE}/external/ggml/include" "${POCKET_SOURCE}/src/models/pocket_tts"
      "${POCKET_BUILD}/generated")
    target_sources(aii_pocket_session PRIVATE $<TARGET_OBJECTS:aii_pocket_history>)
  endif()
  endif()
  add_executable(aii_session_probe probe.cpp native_models.cpp)
  if(NOT DEFINED AII_VAD_LIBRARY)
    set(AII_VAD_LIBRARY "${AII_COMPONENT_BUILD}/vad/libaii_native_vad.dylib")
  endif()
  if(NOT DEFINED AII_ENDPOINT_LIBRARY)
    set(AII_ENDPOINT_LIBRARY "${AII_COMPONENT_BUILD}/endpoint/libaii_native_endpoint.dylib")
  endif()
  foreach(required AII_ASR_LIBRARY AII_VAD_LIBRARY AII_ENDPOINT_LIBRARY)
    if(NOT EXISTS "${${required}}")
      message(FATAL_ERROR "Qualify/build ${required} before configuring the real-model composition")
    endif()
  endforeach()

  target_link_libraries(aii_model_dependencies INTERFACE aii_native_session aii_pocket_session
