cmake_minimum_required(VERSION 3.22)
project(aii_native_endpoint LANGUAGES C CXX)
option(AII_ENDPOINT_ATEN "Candidate native ATen Windows frontend; requires separate numerical qualification" OFF)
option(AII_ENDPOINT_VECTOR_FRONTEND "Candidate ARM SIMD frontend; requires exact paired numerical qualification" OFF)
option(AII_ENDPOINT_SPARSE_FRONTEND "Candidate exact-zero mel pruning; requires paired numerical and full-session qualification" OFF)
if(AII_ENDPOINT_SPARSE_FRONTEND AND AII_ENDPOINT_ATEN)
  message(FATAL_ERROR "Sparse and ATen frontend candidates are mutually exclusive")
endif()
set(AII_ENDPOINT_DEPENDENCIES ORT_INCLUDE ORT_LIBRARY)
if(AII_ENDPOINT_ATEN)
  if(NOT WIN32)
    message(FATAL_ERROR "ATen candidate has only a Windows qualification path")
  endif()
  list(APPEND AII_ENDPOINT_DEPENDENCIES ATEN_ROOT)
else()
  list(APPEND AII_ENDPOINT_DEPENDENCIES POCKETFFT_ROOT SLEEF_INCLUDE SLEEF_LIBRARY)
endif()
foreach(required ${AII_ENDPOINT_DEPENDENCIES})
  if(NOT DEFINED ${required})
    message(FATAL_ERROR "Explicit ${required} required")
  endif()
endforeach()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(aii_native_endpoint SHARED endpoint.cpp frontend.cpp)
target_compile_definitions(aii_native_endpoint PRIVATE AII_ENDPOINT_BUILD)
if(AII_ENDPOINT_SPARSE_FRONTEND)
  target_compile_definitions(aii_native_endpoint PRIVATE AII_ENDPOINT_SPARSE_FRONTEND)
endif()
if(AII_ENDPOINT_VECTOR_FRONTEND)
  if(AII_ENDPOINT_ATEN)
    message(FATAL_ERROR "Vector and ATen frontend candidates are mutually exclusive")
  endif()
  target_compile_definitions(aii_native_endpoint PRIVATE AII_ENDPOINT_VECTOR_FRONTEND)
endif()
target_include_directories(aii_native_endpoint PRIVATE "${ORT_INCLUDE}")
target_link_libraries(aii_native_endpoint PRIVATE "${ORT_LIBRARY}")
if(AII_ENDPOINT_ATEN)
  target_compile_definitions(aii_native_endpoint PRIVATE AII_ENDPOINT_ATEN)
  # Use ordinary includes: CMake's MSVC /external:I form preserves mixed path
  # separators in __FILE__, which defeats the compiler's source-path mapping.
  # This changes warning classification only, not the bound header bytes.
  target_include_directories(aii_native_endpoint PRIVATE "${ATEN_ROOT}/include")
  target_link_libraries(aii_native_endpoint PRIVATE "${ATEN_ROOT}/lib/torch_cpu.lib" "${ATEN_ROOT}/lib/c10.lib")
else()
  add_library(aii_endpoint_fft STATIC fft.cpp)
  set_target_properties(aii_endpoint_fft PROPERTIES POSITION_INDEPENDENT_CODE ON)
  target_include_directories(aii_endpoint_fft PRIVATE "${POCKETFFT_ROOT}")
  target_sources(aii_native_endpoint PRIVATE math.cpp)
  target_include_directories(aii_native_endpoint PRIVATE "${SLEEF_INCLUDE}")
  target_link_libraries(aii_native_endpoint PRIVATE aii_endpoint_fft "${SLEEF_LIBRARY}")
endif()
add_executable(aii_endpoint_probe probe.cpp)
target_link_libraries(aii_endpoint_probe PRIVATE aii_native_endpoint)
add_executable(aii_pause_gate_test pause_gate_test.cpp)
enable_testing()
add_test(NAME pause_audio_clock COMMAND aii_pause_gate_test)
get_filename_component(ORT_ROOT "${ORT_LIBRARY}" DIRECTORY)
set_target_properties(aii_native_endpoint PROPERTIES BUILD_RPATH "${ORT_ROOT}")
if(MSVC)
  # The bound SLEEF build is static; Windows headers otherwise request a DLL
  # import thunk that the static library cannot satisfy.
  target_compile_definitions(aii_native_endpoint PRIVATE SLEEF_STATIC_LIBS)
  target_compile_options(aii_native_endpoint PRIVATE /W4 /fp:strict)
else()
  target_compile_options(aii_native_endpoint PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
endif()
