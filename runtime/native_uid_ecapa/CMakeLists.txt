cmake_minimum_required(VERSION 3.22)
project(aii_ecapa_frontend LANGUAGES CXX)
include(../cmake/SourcePrivacy.cmake)
find_package(Threads REQUIRED)
if(NOT DEFINED POCKETFFT_ROOT)
  message(FATAL_ERROR "Explicit POCKETFFT_ROOT required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(AII_ECAPA_STATIC "Link the frontend into its native owner" OFF)
if(AII_ECAPA_STATIC)
  add_library(aii_ecapa_frontend STATIC frontend.cpp)
  target_compile_definitions(aii_ecapa_frontend PUBLIC AII_ECAPA_STATIC)
  set_target_properties(aii_ecapa_frontend PROPERTIES POSITION_INDEPENDENT_CODE ON)
else()
  add_library(aii_ecapa_frontend SHARED frontend.cpp)
endif()
target_compile_definitions(aii_ecapa_frontend PRIVATE AII_ECAPA_BUILD)
set_target_properties(aii_ecapa_frontend PROPERTIES
  CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
target_include_directories(aii_ecapa_frontend PRIVATE "${POCKETFFT_ROOT}")
add_executable(aii_ecapa_frontend_test contract_test.cpp)
target_link_libraries(aii_ecapa_frontend_test PRIVATE aii_ecapa_frontend Threads::Threads)
foreach(target aii_ecapa_frontend aii_ecapa_frontend_test)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /fp:strict)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
  endif()
endforeach()
enable_testing()
add_test(NAME ecapa_frontend_contracts COMMAND aii_ecapa_frontend_test)
