cmake_minimum_required(VERSION 3.22)
project(aii_native_asr LANGUAGES C CXX)
enable_testing()
add_executable(aii_asr_execution_policy_test execution_policy_test.cpp ../native/vendor/cjson/cJSON.c)
target_include_directories(aii_asr_execution_policy_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../native/vendor/cjson")
target_compile_features(aii_asr_execution_policy_test PRIVATE cxx_std_17)
add_test(NAME asr_execution_policy_contracts COMMAND aii_asr_execution_policy_test)
foreach(required ORT_INCLUDE ORT_LIBRARY)
  if(NOT DEFINED ${required})
    message(FATAL_ERROR "Explicit bound ${required} required")
  endif()
endforeach()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_executable(aii_onnx_log_test onnx_log_test.cpp)
target_include_directories(aii_onnx_log_test PRIVATE "${ORT_INCLUDE}")
add_test(NAME native_onnx_diagnostic_policy COMMAND aii_onnx_log_test)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/../native/vendor/picosha2/picosha2.h" sha_digest)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/../native/vendor/cjson/cJSON.c" json_digest)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/../native/vendor/cjson/cJSON.h" json_header_digest)
if(NOT sha_digest STREQUAL "b13c180161ffac8d0adc81e033e493c409457c4d1258ab9781ac80579ba3bdd8" OR
   NOT json_digest STREQUAL "607e756460fa0de37d20a7a9181f2de29c97bfb7ce5a0e6c2f548243836cd852" OR
   NOT json_header_digest STREQUAL "25b0145150d500498e4d209cec69c18c42cf818bffcc54690be3b895a2a16dee")
  message(FATAL_ERROR "Bound native asset dependencies changed")
endif()
add_library(aii_native_asr SHARED asr.cpp frontend.cpp initializers.cpp
  ../native/platform/readonly_model.cpp ../native/vendor/cjson/cJSON.c)
target_include_directories(aii_native_asr PRIVATE "${ORT_INCLUDE}" "${CMAKE_CURRENT_SOURCE_DIR}/../native/vendor/cjson")
target_link_libraries(aii_native_asr PRIVATE "${ORT_LIBRARY}")
option(AII_ASR_DIRECTML "Build explicit Windows DirectML encoder selection" OFF)
if(AII_ASR_DIRECTML)
  if(NOT WIN32)
    message(FATAL_ERROR "DirectML is a Windows target, never an implicit substitute")
  endif()
  target_compile_definitions(aii_native_asr PRIVATE AII_ASR_DIRECTML NOMINMAX)
  target_link_libraries(aii_native_asr PRIVATE dxgi d3d12)
endif()
if(WIN32)
  target_link_libraries(aii_native_asr PRIVATE bcrypt)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  # The bound iOS ORT framework is static; its Apple dependencies belong in
  # the app-linked build, not in a fetched desktop runtime.
  target_link_libraries(aii_native_asr PRIVATE "-framework Foundation"
    "-framework CoreML" "-framework Accelerate")
endif()
if(MSVC)
  target_compile_options(aii_native_asr PRIVATE /W4 /fp:strict)
else()
  target_compile_options(aii_native_asr PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
endif()
get_filename_component(ORT_ROOT "${ORT_LIBRARY}" DIRECTORY)
set_target_properties(aii_native_asr PROPERTIES BUILD_RPATH "${ORT_ROOT}")
add_executable(aii_asr_probe probe.cpp frontend.cpp)
target_link_libraries(aii_asr_probe PRIVATE aii_native_asr)
add_executable(aii_asr_frontend_test frontend_test.cpp frontend.cpp)
add_test(NAME frontend_contracts COMMAND aii_asr_frontend_test)
add_executable(aii_asr_initializers_test initializers_test.cpp initializers.cpp
  ../native/platform/readonly_model.cpp ../native/vendor/cjson/cJSON.c)
target_include_directories(aii_asr_initializers_test PRIVATE "${ORT_INCLUDE}")
target_link_libraries(aii_asr_initializers_test PRIVATE "${ORT_LIBRARY}")
option(AII_LINUX_NATIVE_SHA256 "Explicit Linux desktop accelerated integrity candidate" OFF)
if(AII_LINUX_NATIVE_SHA256)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR ANDROID)
    message(FATAL_ERROR "Native OpenSSL SHA256 candidate is Linux desktop only")
  endif()
  find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
  foreach(target aii_native_asr aii_asr_initializers_test)
    target_compile_definitions(${target} PRIVATE AII_LINUX_NATIVE_SHA256)
    target_link_libraries(${target} PRIVATE OpenSSL::Crypto)
  endforeach()
endif()
set_target_properties(aii_asr_initializers_test PROPERTIES BUILD_RPATH "${ORT_ROOT}")
if(WIN32)
  target_link_libraries(aii_asr_initializers_test PRIVATE bcrypt)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  target_link_libraries(aii_asr_initializers_test PRIVATE "-framework Foundation"
    "-framework CoreML" "-framework Accelerate")
endif()
add_test(NAME sealed_initializer_contracts COMMAND aii_asr_initializers_test)
set_tests_properties(sealed_initializer_contracts PROPERTIES TIMEOUT 30)
if(MSVC)
  target_compile_options(aii_asr_initializers_test PRIVATE /W4 /fp:strict)
else()
  target_compile_options(aii_asr_initializers_test PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
endif()
if(NOT MSVC)
  target_compile_options(aii_asr_probe PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
  target_compile_options(aii_asr_frontend_test PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
else()
  target_compile_options(aii_asr_probe PRIVATE /W4 /fp:strict)
  target_compile_options(aii_asr_frontend_test PRIVATE /W4 /fp:strict)
endif()
