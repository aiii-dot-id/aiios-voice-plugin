cmake_minimum_required(VERSION 3.18)
project(uid_pixel_probe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT ANDROID OR NOT UID_NATIVE_ROOT OR NOT UID_ORT_ROOT)
  message(FATAL_ERROR "Android NDK and explicit verified source/runtime roots required")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(UID_KNF_ROOT "${UID_NATIVE_ROOT}/knf")
set(UID_KISS_ROOT "${UID_NATIVE_ROOT}/kiss")
add_subdirectory("${UID_NATIVE_ROOT}" native)
add_library(ort SHARED IMPORTED)
set_target_properties(ort PROPERTIES IMPORTED_LOCATION "${UID_ORT_ROOT}/jni/arm64-v8a/libonnxruntime.so")
add_executable(uid_pixel_probe main.cc)
target_include_directories(uid_pixel_probe PRIVATE "${UID_NATIVE_ROOT}" "${UID_ORT_ROOT}/headers")
target_link_libraries(uid_pixel_probe PRIVATE aiii_uid_frontend ort neuralnetworks vulkan)
target_compile_options(uid_pixel_probe PRIVATE -fno-fast-math -ffp-contract=on -Wall -Wextra)
