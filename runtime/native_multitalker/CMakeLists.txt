      # ELF symbol interposition can silently bind the diarizer to a different
      # ggml embedded in TTS. Require its ggml implementation to remain private.
      execute_process(COMMAND "${CMAKE_NM}" -D -P
        "${NEMOTRON_ROOT}/lib/libnemo_speech_asr.so"
        RESULT_VARIABLE nm_status OUTPUT_VARIABLE nemo_symbols ERROR_VARIABLE nm_error)
      if(NOT nm_status EQUAL 0 OR nemo_symbols MATCHES "(^|\n)ggml_")
        message(FATAL_ERROR "Nemotron requires private static ggml linkage; see NEMOTRON_NATIVE_DESKTOPS.md")
      endif()
    endif()
    target_sources(aii_multitalker_onnx PRIVATE nemotron.cpp)
    target_include_directories(aii_multitalker_onnx PRIVATE "${NEMOTRON_ROOT}/include")
