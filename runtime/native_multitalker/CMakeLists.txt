cmake_minimum_required(VERSION 3.22)
project(aii_native_multitalker LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(../cmake/SourcePrivacy.cmake)
enable_testing()
find_package(Threads REQUIRED)
add_library(aii_multitalker_decoder STATIC decoder.cpp)
add_executable(aii_multitalker_decoder_test decoder_test.cpp)
target_link_libraries(aii_multitalker_decoder_test PRIVATE aii_multitalker_decoder Threads::Threads)
add_test(NAME multitalker_decoder_contracts COMMAND aii_multitalker_decoder_test)
set_tests_properties(multitalker_decoder_contracts PROPERTIES TIMEOUT 10)
if(DEFINED ORT_INCLUDE AND DEFINED ORT_LIBRARY)
  add_library(aii_multitalker_onnx STATIC onnx_backend.cpp)
  target_include_directories(aii_multitalker_onnx PRIVATE "${ORT_INCLUDE}")
  target_link_libraries(aii_multitalker_onnx PUBLIC aii_multitalker_decoder "${ORT_LIBRARY}")
  add_executable(aii_multitalker_trace_probe trace_probe.cpp)
  target_link_libraries(aii_multitalker_trace_probe PRIVATE aii_multitalker_onnx)
endif()
foreach(target aii_multitalker_decoder aii_multitalker_decoder_test aii_multitalker_onnx aii_multitalker_trace_probe)
  if(TARGET ${target})
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /WX /fp:strict)
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
    endif()
  endif()
endforeach()
