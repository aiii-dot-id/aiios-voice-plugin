cmake_minimum_required(VERSION 3.22)
project(aii_native_uid LANGUAGES C CXX)
option(AII_UID_NCNN "Explicit native GPU UID representation; no ORT/CPU fallback" OFF)
set(uid_required UID_FRONTEND_SOURCE UID_KNF_ROOT UID_KISS_ROOT)
if(AII_UID_NCNN)
  list(APPEND uid_required ncnn_DIR)
else()
  list(APPEND uid_required ORT_INCLUDE ORT_LIBRARY)
endif()
foreach(required ${uid_required})
  if(NOT DEFINED ${required})
    message(FATAL_ERROR "Explicit bound ${required} required")
  endif()
endforeach()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS ON)
add_subdirectory("${UID_FRONTEND_SOURCE}" frontend)
add_library(aii_native_uid SHARED uid.cpp)
target_compile_definitions(aii_native_uid PRIVATE AII_UID_BUILD)
target_include_directories(aii_native_uid PRIVATE "${UID_FRONTEND_SOURCE}")
target_link_libraries(aii_native_uid PRIVATE aiii_uid_frontend)
if(AII_UID_NCNN)
  find_package(ncnn REQUIRED)
  target_compile_definitions(aii_native_uid PRIVATE AII_UID_NCNN)
  target_link_libraries(aii_native_uid PRIVATE ncnn)
  # ncnn was compiled without exceptions; this private C ABI catches its own
  # validation exceptions and never throws through upstream callbacks.
  set_source_files_properties(uid.cpp PROPERTIES COMPILE_OPTIONS "-fexceptions")
else()
  target_include_directories(aii_native_uid PRIVATE "${ORT_INCLUDE}")
  target_link_libraries(aii_native_uid PRIVATE "${ORT_LIBRARY}")
endif()
add_executable(aii_uid_probe probe.cpp)
target_link_libraries(aii_uid_probe PRIVATE aii_native_uid)
if(NOT AII_UID_NCNN)
  add_executable(aii_uid_contract_test contract_test.cpp)
  target_include_directories(aii_uid_contract_test PRIVATE "${ORT_INCLUDE}")
endif()
enable_testing()
add_executable(aii_uid_model_contract_test model_contract_test.cpp)
add_test(NAME uid_exact_model_binding COMMAND aii_uid_model_contract_test)
if(NOT AII_UID_NCNN)
  add_test(NAME uid_cancel_fault_classification COMMAND aii_uid_contract_test)
endif()
get_filename_component(ORT_ROOT "${ORT_LIBRARY}" DIRECTORY)
set_target_properties(aii_native_uid PROPERTIES BUILD_RPATH "${ORT_ROOT}")
foreach(target aii_native_uid aii_uid_probe aii_uid_contract_test)
  if(NOT TARGET ${target})
    continue()
  endif()
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /fp:strict)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
  endif()
endforeach()
