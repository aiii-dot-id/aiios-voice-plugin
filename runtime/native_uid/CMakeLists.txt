cmake_minimum_required(VERSION 3.22)
project(aii_native_uid LANGUAGES C CXX)
include(../cmake/SourcePrivacy.cmake)
option(AII_UID_NCNN "Explicit native GPU UID representation; no ORT/CPU fallback" OFF)
option(AII_UID_COREML "Enable explicitly requested production CoreML UID; no profiling prerequisite" OFF)
if(AII_UID_COREML AND (NOT APPLE OR AII_UID_NCNN))
  message(FATAL_ERROR "CoreML UID requires an Apple ORT build")
endif()
set(uid_required UID_FRONTEND_SOURCE UID_KNF_ROOT UID_KISS_ROOT)
if(AII_UID_NCNN)
  list(APPEND uid_required ncnn_DIR)
else()
  list(APPEND uid_required ORT_INCLUDE ORT_LIBRARY)
endif()
foreach(required ${uid_required})
  if(NOT DEFINED ${required})
    message(FATAL_ERROR "Explicit bound ${required} required")
  endif()
endforeach()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS ON)
add_subdirectory("${UID_FRONTEND_SOURCE}" frontend)
add_library(aii_native_uid SHARED uid.cpp)
if(AII_UID_COREML)
  target_compile_definitions(aii_native_uid PRIVATE AII_UID_COREML)
  add_executable(aii_uid_coreml_cache coreml_cache.cpp)
  target_compile_definitions(aii_uid_coreml_cache PRIVATE AII_UID_COREML)
  target_compile_features(aii_uid_coreml_cache PRIVATE cxx_std_17)
  target_include_directories(aii_uid_coreml_cache PRIVATE "${ORT_INCLUDE}")
  target_link_libraries(aii_uid_coreml_cache PRIVATE "${ORT_LIBRARY}")
endif()
option(AII_UID_QUALIFICATION_PROFILE "Development-only ORT node-placement profiling; never package" OFF)
if(AII_UID_QUALIFICATION_PROFILE)
  if(AII_UID_NCNN)
    message(FATAL_ERROR "ORT qualification profiling requires an ORT build")
  endif()
  target_compile_definitions(aii_native_uid PRIVATE AII_UID_QUALIFICATION_PROFILE)
endif()
option(AII_UID_ECAPA "Include exact ECAPA frontend/model candidate" OFF)
if(AII_UID_ECAPA)
  if(AII_UID_NCNN)
    message(FATAL_ERROR "ECAPA has no qualified ncnn representation")
  endif()
  # Keep the exact frontend inside the UID component. A candidate must not
  # acquire an undeclared runtime library dependency during packaging.
  set(AII_ECAPA_STATIC ON)
  add_subdirectory(../native_uid_ecapa ecapa)
  target_compile_definitions(aii_native_uid PRIVATE AII_UID_ECAPA)
  target_link_libraries(aii_native_uid PRIVATE aii_ecapa_frontend)
  enable_language(C)
  find_package(Threads REQUIRED)
  add_executable(aii_ecapa_lifecycle_probe ecapa_lifecycle_probe.cpp
    identity.cpp snapshot.cpp enrollment.cpp speaker_registry.cpp
    ../native/session/native_speaker.cpp ../native/platform/readonly_model.cpp
    ../native/vendor/cjson/cJSON.c)
  target_compile_features(aii_ecapa_lifecycle_probe PRIVATE cxx_std_17)
  target_include_directories(aii_ecapa_lifecycle_probe PRIVATE ../native/vendor/cjson)
  target_link_libraries(aii_ecapa_lifecycle_probe PRIVATE aii_native_uid Threads::Threads ${CMAKE_DL_LIBS})
  if(WIN32)
    target_link_libraries(aii_ecapa_lifecycle_probe PRIVATE bcrypt)
  endif()
endif()
target_compile_definitions(aii_native_uid PRIVATE AII_UID_BUILD)
target_include_directories(aii_native_uid PRIVATE "${UID_FRONTEND_SOURCE}")
target_link_libraries(aii_native_uid PRIVATE aiii_uid_frontend)
if(AII_UID_NCNN)
  find_package(ncnn REQUIRED)
  target_compile_definitions(aii_native_uid PRIVATE AII_UID_NCNN)
  target_link_libraries(aii_native_uid PRIVATE ncnn)
  # ncnn was compiled without exceptions; this private C ABI catches its own
  # validation exceptions and never throws through upstream callbacks.
  set_source_files_properties(uid.cpp PROPERTIES COMPILE_OPTIONS "-fexceptions")
else()
  target_include_directories(aii_native_uid PRIVATE "${ORT_INCLUDE}")
  target_link_libraries(aii_native_uid PRIVATE "${ORT_LIBRARY}")
endif()
add_executable(aii_uid_probe probe.cpp)
target_link_libraries(aii_uid_probe PRIVATE aii_native_uid)
if(NOT AII_UID_NCNN)
  add_executable(aii_uid_contract_test contract_test.cpp)
  target_include_directories(aii_uid_contract_test PRIVATE "${ORT_INCLUDE}")
endif()
enable_testing()
add_executable(aii_uid_model_contract_test model_contract_test.cpp)
add_test(NAME uid_exact_model_binding COMMAND aii_uid_model_contract_test)
if(NOT AII_UID_NCNN)
  add_test(NAME uid_cancel_fault_classification COMMAND aii_uid_contract_test)
endif()
get_filename_component(ORT_ROOT "${ORT_LIBRARY}" DIRECTORY)
set_target_properties(aii_native_uid PROPERTIES BUILD_RPATH "${ORT_ROOT}")
foreach(target aii_native_uid aii_uid_probe aii_uid_contract_test)
  if(NOT TARGET ${target})
    continue()
  endif()
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /fp:strict)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
  endif()
endforeach()
