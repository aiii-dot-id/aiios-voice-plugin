cmake_minimum_required(VERSION 3.22)
project(aii_native_vad LANGUAGES CXX)
foreach(required ORT_INCLUDE ORT_LIBRARY)
  if(NOT DEFINED ${required})
    message(FATAL_ERROR "Explicit bound ${required} required")
  endif()
endforeach()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(aii_native_vad SHARED vad.cpp)
target_compile_definitions(aii_native_vad PRIVATE AII_VAD_BUILD)
target_include_directories(aii_native_vad PRIVATE "${ORT_INCLUDE}")
target_link_libraries(aii_native_vad PRIVATE "${ORT_LIBRARY}")
add_executable(aii_vad_probe probe.cpp)
target_link_libraries(aii_vad_probe PRIVATE aii_native_vad)
get_filename_component(ORT_ROOT "${ORT_LIBRARY}" DIRECTORY)
set_target_properties(aii_native_vad PROPERTIES BUILD_RPATH "${ORT_ROOT}")
foreach(target aii_native_vad aii_vad_probe)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /fp:strict)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
  endif()
endforeach()
