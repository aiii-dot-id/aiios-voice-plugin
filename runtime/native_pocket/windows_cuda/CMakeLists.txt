cmake_minimum_required(VERSION 3.24)
project(native_cuda_candidate LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_CUDA_ARCHITECTURES STREQUAL "61-real")
  message(FATAL_ERROR "Private GTX 1070 Windows build requires explicit 61-real")
endif()
foreach(required ENGINE_SOURCE OVERRIDES)
  if(NOT DEFINED ${required})
    message(FATAL_ERROR "Missing ${required}")
  endif()
endforeach()
set(ENGINE_ENABLE_CUDA ON CACHE BOOL "" FORCE)
set(ENGINE_ENABLE_CUDA_GRAPHS ON CACHE BOOL "" FORCE)
set(ENGINE_ENABLE_VULKAN OFF CACHE BOOL "" FORCE)
set(ENGINE_ENABLE_METAL OFF CACHE BOOL "" FORCE)
set(ENGINE_ENABLE_HIP OFF CACHE BOOL "" FORCE)
set(ENGINE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ENGINE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(AUDIOCPP_MODEL_SET custom CACHE STRING "" FORCE)
set(AUDIOCPP_MODELS pocket_tts CACHE STRING "" FORCE)
set(AUDIOCPP_BUILD_NATIVE_MODEL_MANAGER OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_CUDA_RUNTIME_LIBRARY Shared)
add_subdirectory("${ENGINE_SOURCE}" engine EXCLUDE_FROM_ALL)
add_library(native_pocket_resident SHARED
  "${OVERRIDES}/resident.cpp" "${OVERRIDES}/mimi_decoder.cpp"
  "${OVERRIDES}/acoustic_model.cpp" "${OVERRIDES}/assets_bound.cpp"
  "${OVERRIDES}/session.cpp" "${OVERRIDES}/profiling.cpp"
  "${OVERRIDES}/resource_bundle.cpp" "${OVERRIDES}/filesystem.cpp"
  "${OVERRIDES}/safetensors.cpp")
target_compile_features(native_pocket_resident PRIVATE cxx_std_17)
target_compile_options(native_pocket_resident PRIVATE /O2 /EHsc /openmp)
target_compile_definitions(native_pocket_resident PRIVATE GGML_GELU_FORCE_F32)
target_include_directories(native_pocket_resident PRIVATE "${OVERRIDES}"
  "${ENGINE_SOURCE}/src/models/pocket_tts")
target_link_libraries(native_pocket_resident PRIVATE engine_runtime)
add_executable(cuda_speech_probe "${CMAKE_CURRENT_LIST_DIR}/thread_profile.cpp")
target_compile_features(cuda_speech_probe PRIVATE cxx_std_17)
target_link_libraries(cuda_speech_probe PRIVATE native_pocket_resident psapi)
